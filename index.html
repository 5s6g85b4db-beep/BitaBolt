УДАЛЕНИЕ ПОЛЬЗОВАТЕЛЯ (только админ) ---- async function deleteUserAccount(username) { if (S.me?.un !== 'admin') { toast('❌ Только администратор может удалять аккаунты', '⛔'); return; } if (username === 'admin') { toast('❌ Нельзя удалить администратора', '⛔'); return; } try { // Удаляем из Firebase await remove(ref(db, 'users/' + username)); await remove(ref(db, 'usernames/' + username)); // Удаляем из локального хранилища const all = accounts(); delete all[username]; Store.set('bb_accounts', all); // Удаляем из памяти delete S.users[username]; // Удаляем сообщения пользователя Object.keys(S.msgs).forEach(chatId => { S.msgs[chatId] = S.msgs[chatId].filter(m => m.u !== username && m.u !== 'me'); }); persist(); renderList(); toast('✅ Пользователь @' + username + ' удалён', '🗑️'); } catch (e) { console.error('Delete error:', e); toast('❌ Ошибка удаления: ' + e.message, '❌'); } } // ---- 4.
